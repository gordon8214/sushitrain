cmake_minimum_required(VERSION 3.22)
project(SynctrainKDE VERSION 2.8 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(QT_MAJOR_VERSION 6)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
option(BUILD_SHARED_LIBS "Build loadable KDE plugins" ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network Qml Quick QuickControls2 Multimedia Pdf Test)
find_package(KF6 REQUIRED COMPONENTS Kirigami I18n CoreAddons DBusAddons KIO Notifications StatusNotifierItem Prison)
qt_standard_project_setup(REQUIRES 6.8)
add_library(synctrain-desktop-client STATIC src/serviceclient.cpp src/serviceclient.h)
set_target_properties(synctrain-desktop-client PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(synctrain-desktop-client PUBLIC Qt6::Core Qt6::Network)
add_executable(synctrain-kde src/main.cpp src/backend.cpp src/management.cpp src/remote.cpp src/browser.cpp src/backend.h)
target_link_libraries(synctrain-kde PRIVATE synctrain-desktop-client Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Pdf KF6::Kirigami KF6::I18n KF6::CoreAddons KF6::DBusAddons KF6::KIOFileWidgets KF6::Notifications KF6::StatusNotifierItem KF6::Prison)
qt_add_qml_module(synctrain-kde URI com.gordonbyrnes.synctrain VERSION 1.0 QML_FILES qml/Main.qml qml/Preview.qml)
install(TARGETS synctrain-kde RUNTIME DESTINATION ${KDE_INSTALL_BINDIR})
add_subdirectory(plugins)
if(BUILD_TESTING)
 add_executable(kde-backend-test tests/backendtest.cpp src/backend.cpp src/management.cpp src/remote.cpp src/browser.cpp src/backend.h)
 target_link_libraries(kde-backend-test PRIVATE Qt6::Test synctrain-desktop-client Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Pdf KF6::Kirigami KF6::I18n KF6::CoreAddons KF6::DBusAddons KF6::KIOFileWidgets KF6::Notifications KF6::StatusNotifierItem KF6::Prison)
 qt_add_resources(kde-backend-test "test-ui" PREFIX "/test-ui" FILES qml/Main.qml qml/Preview.qml)
 add_test(NAME kde-backend COMMAND kde-backend-test)
 set_tests_properties(kde-backend PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 add_executable(kde-client-test tests/clienttest.cpp)
 target_link_libraries(kde-client-test PRIVATE synctrain-desktop-client Qt6::Test)
 add_test(NAME kde-client COMMAND kde-client-test)
 set_tests_properties(kde-client PROPERTIES TIMEOUT 60)
endif()
target_compile_options(synctrain-desktop-client PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(synctrain-kde PRIVATE -Wall -Wextra -Wpedantic)
